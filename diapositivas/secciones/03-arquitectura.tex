\begin{frame}[t]{Tres decisiones que explican el resto de la charla}
  \note{Estas tres son las que hay que saber defender. Si solo queda tiempo
  para una, la primera: es la que demuestra que la regla crítica no depende
  de que el código se porte bien. El registro completo de decisiones está en
  el informe (E1, §4).}

  \footnotesize
  \decisionslide{01}{Solapamiento resuelto en la base de datos}%
    {Restricción \id{EXCLUDE} sobre \id{tstzrange} en \id{reservas\_db},
     además de la comprobación en el servicio.}%
    {Confiar solo en la validación aplicativa: bajo peticiones concurrentes
     deja pasar la doble reserva.}

  \decisionslide{02}{Una base por microservicio, con rol propio}%
    {Tres bases aisladas por credenciales; ningún servicio lee las tablas de
     otro aunque quiera.}%
    {Esquema compartido: convierte la independencia de datos en una promesa
     que nadie verifica.}

  \decisionslide{03}{Un \id{edge} que unifica el origen}%
    {Nginx sirve shell, remotes y \id{/api} bajo un mismo origen.}%
    {CORS servicio por servicio: filtra los puertos internos al navegador.}
\end{frame}
