% --- Pruebas ----------------------------------------------------------------
\begin{frame}{Estrategia de pruebas}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}\footnotesize
        \item \textbf{73 métodos de prueba} en 12 clases.
        \item \textbf{Unitarias de dominio y aplicación} sobre las reglas:
              \id{BookingServiceCreacionTest},
              \id{BookingServiceCancelacionTest}, \id{CourtServiceTest},
              \id{ReportServiceTest}, \id{UserServiceTest}.
        \item \textbf{De integración con PostgreSQL real}:
              \id{ConcurrenciaReservaIT}, que es la única forma de probar
              \rn{02} bajo concurrencia.
        \item \textbf{Del gateway}: \id{AuthenticationFilterTest}, sobre la
              propagación de identidad y rol.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Por qué \id{ConcurrenciaReservaIT} repite}
        \footnotesize
        Una carrera que se gana una vez puede ser suerte. La prueba lanza las
        dos peticiones simultáneas \textbf{diez veces seguidas} y exige que en
        las diez gane exactamente una.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% --- Antes de la demo -------------------------------------------------------
\begin{frame}{Lista de comprobación previa a la demo}
  \note{Para el ensayo, no para el aula. Existe porque casi todo lo que falla
  en una demo, falla por algo de esta lista.}

  \begin{itemize}\footnotesize
    \item \id{docker compose up -d --wait} lanzado \alert{antes} de entrar al
          aula; \id{docker compose ps} con los once servicios en
          \id{healthy}.
    \item Dos navegadores (o uno normal y uno privado) con sesión iniciada:
          uno como socio, otro como administrador.
    \item Datos recién sembrados: \id{docker compose down -v \&\& docker
          compose up -d --wait}, para que los reportes no arrastren reservas
          viejas.
    \item Zoom del navegador al 125\,\%: lo que se lee en el portátil no se
          lee en el proyector.
    \item Notificaciones del sistema operativo silenciadas.
    \item \textbf{Plan B:} un vídeo corto de la demo, ya abierto en el
          escritorio.
  \end{itemize}
\end{frame}

% --- Preguntas --------------------------------------------------------------
\begin{frame}{Preguntas que esperamos}
  \note{Ensayarlas en voz alta. Una respuesta pensada en el momento suena a
  que no se pensó nunca.}

  \begin{itemize}\footnotesize
    \item ¿Por qué microservicios para un sistema de este tamaño?
    \item ¿Qué pasa si \id{ms-canchas} está caído cuando alguien reserva?
          \emph{(Devuelve 503: no se crea una reserva sobre una cancha que no
          se pudo validar.)}
    \item ¿Por qué no una cola de mensajes entre servicios?
    \item ¿Module Federation no acopla los remotes al shell por la versión de
          React? \emph{(Sí: por eso React va como \id{singleton} declarado en
          los cuatro.)}
    \item ¿Cómo evitáis que un remote roto tumbe toda la aplicación?
    \item ¿Cuánto de esto lo escribió el modelo y cuánto vosotros?
  \end{itemize}
\end{frame}
