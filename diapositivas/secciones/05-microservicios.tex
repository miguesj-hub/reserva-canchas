       {Arquitectura de microservicios (15\,\%)}

\begin{frame}{Cuatro servicios Spring Boot y un gateway}
  \note{Leer la tabla de izquierda a derecha una sola vez. Si preguntan por
  qué ms-reportes no tiene base: porque compone por HTTP lo que ya está en
  los otros dos, y un almacén analítico sería más arquitectura de la que
  este alcance necesita.}

  \begin{center}
    \scriptsize
    \setlength{\tabcolsep}{4pt}
    \renewcommand{\arraystretch}{1.05}
    \begin{tabular}{@{}llL{5.6cm}@{}}
      \toprule
      \textbf{Servicio} & \textbf{Base} & \textbf{Responsabilidad}\\
      \midrule
      \id{api-gateway} & ---               & Entrada única: autentica y propaga identidad y rol\\
      \id{ms-usuarios} & \id{usuarios\_db} & Registro, autenticación, usuarios y roles\\
      \id{ms-canchas}  & \id{canchas\_db}  & Catálogo, horarios de atención y bloqueos\\
      \id{ms-reservas} & \id{reservas\_db} & Disponibilidad, creación y cancelación\\
      \id{ms-reportes} & \alert{sin base}  & Los cuatro indicadores, componiendo por REST\\
      \bottomrule
    \end{tabular}
  \end{center}

  \vspace{2pt}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{itemize}\scriptsize
        \item \textbf{Contrato antes que código:} 4 documentos OpenAPI, 20
              rutas, escritos antes del primer \emph{endpoint}. Publicados en
              \id{/swagger-ui}.
        \item Hexagonal: el dominio no conoce ni HTTP ni JPA.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      {\scriptsize \textbf{Quién decide quién eres:} el gateway valida la
      credencial contra \id{ms-usuarios} y escribe \id{X-User-Id} y
      \id{X-User-Role}. Los servicios confían en esas cabeceras
      \emph{porque nadie más puede escribirlas}.}
    \end{column}
  \end{columns}
\end{frame}
