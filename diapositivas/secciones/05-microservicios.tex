\begin{frame}{Cuatro servicios y un gateway}

  \begin{center}
    \footnotesize
    \begin{tabular}{@{}llp{5cm}@{}}
      \toprule
      \textbf{Servicio} & \textbf{Base} & \textbf{Responsabilidad}\\
      \midrule
      \id{api-gateway} & --- & Entrada única; verifica la credencial y rutea\\
      \id{ms-usuarios} & \id{usuarios\_db} & Registro, autenticación, usuarios y roles\\
      \id{ms-canchas}  & \id{canchas\_db}  & Catálogo, horarios de atención y bloqueos\\
      \id{ms-reservas} & \id{reservas\_db} & Disponibilidad, creación y cancelación\\
      \id{ms-reportes} & \alert{sin base}  & Indicadores, componiendo datos de los otros dos\\
      \bottomrule
    \end{tabular}
  \end{center}

  \vspace{2pt}
