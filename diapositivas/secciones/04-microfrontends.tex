\bloque{Frontend}{0:50}{\exponeFrontend}%
       {Arquitectura de microfrontends (15\,\%)}

\begin{frame}[fragile]{Un host, tres remotes, desplegables por separado}
  \note{Los remotes se resuelven al cargar la página, no al compilar el
  shell: por eso se puede redesplegar uno sin tocar los otros. La rúbrica
  pide dos; entregamos tres, cada uno con su package.json, su build y su
  imagen.}

  \begin{columns}[T]
    \begin{column}{0.53\textwidth}
      \begin{lstlisting}[language=jsts,basicstyle=\ttfamily\tiny]
pluginModuleFederation({
  name: 'shell',
  remotes: {
    mfReservas:       `mfReservas@${url.reservas}`,
    mfAdministracion: `mfAdministracion@${url.admin}`,
    mfReportes:       `mfReportes@${url.reportes}`,
  },
  shared: {
    react:             { singleton: true },
    'react-dom':       { singleton: true },
    'react-router-dom':{ singleton: true },
  },
})
      \end{lstlisting}
      {\scriptsize\color{textoSuave} \id{frontend/shell/rsbuild.config.ts}}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{itemize}\scriptsize
        \item Cada remote se carga \textbf{por URL, en ejecución}. El shell no
              los compila dentro.
        \item \id{singleton}: React y el enrutador se cargan una sola vez.
        \item El shell inyecta la sesión; el remote no la reimplementa.
      \end{itemize}
      \begin{block}{Desplegables por separado}
        \scriptsize
        Con el sistema en pie:\\
        \id{docker compose up -d}\\
        \id{\ \ \ \ --build mf-reportes}\\[3pt]
        Se recarga el navegador y el cambio está. \alert{El shell y los otros
        dos remotes nunca se reconstruyeron.}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
