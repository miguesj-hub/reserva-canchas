\bloque{El problema y el alcance}{1,5 min}{\exponeArquitectura}%
       {Alcance funcional implementado (25\,\%)}

\begin{frame}{El problema}
  \guion{Una frase de contexto y pasar. El docente escribió el documento de
  alcance: no hay que explicárselo, hay que demostrar que se leyó.}

  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      Un club con canchas de \textbf{pádel, tenis y básquet} necesita que:
      \begin{itemize}
        \item el socio vea qué bloques quedan libres, reserve y cancele;
        \item el administrador gestione el catálogo, cancele ante un
              mantenimiento imprevisto y mida el uso real.
      \end{itemize}

      \vspace{6pt}
      Dos roles, cuatro módulos, ocho reglas de negocio.
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Lo que NO está en alcance}
        \footnotesize
        Pagos en línea · notificaciones · reservas recurrentes · torneos ·
        app móvil nativa · BI avanzado.
      \end{block}
      {\scriptsize\color{textoSuave} Delimitado así en el documento de
      alcance, §3.5. No lo construimos, y es deliberado.}
    \end{column}
  \end{columns}
\end{frame}

  \begin{columns}[T]

    \end{column}
