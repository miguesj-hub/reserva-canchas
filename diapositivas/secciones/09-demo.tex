\begin{frame}{Lo que vais a ver corriendo}
  \note{Proyectar esta lámina mientras se cambia de ventana. El paso 3 es el
  momento importante de toda la charla: el 10 % de reglas de negocio y parte
  del 25 % de alcance, demostrado en diez segundos. Si algo falla, se pasa
  al vídeo de respaldo sin anunciarlo.}

  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{Como socio}
      \begin{enumerate}\scriptsize
        \item Entrar y ver la disponibilidad de una cancha para hoy.
        \item Reservar un bloque libre.
        \item Intentar reservar \alert{el mismo bloque} desde otra sesión:
              el sistema lo rechaza (\rn{02}).
        \item Cancelar la reserva desde «mis reservas».
        \item El bloque vuelve a estar libre (\rn{05}).
      \end{enumerate}
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{Como administrador}
      \begin{enumerate}[start=6]\scriptsize
        \item Crear una cancha y verla en el catálogo del socio.
        \item Cancelar la reserva de otro usuario (\rn{03}).
        \item Abrir reportes: el número refleja lo que acaba de pasar.
      \end{enumerate}
      \vspace{4pt}
      \begin{block}{Si sobra tiempo}
        \scriptsize
        Reconstruir \id{mf-reportes} con el sistema en pie, para enseñar el
        despliegue independiente.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
