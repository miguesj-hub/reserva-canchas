\bloque{Cómo lo construimos}{0:50}{\exponeArquitectura}%
       {Calidad técnica y documentación (10\,\%)}

\begin{frame}{No fue \emph{vibe coding}: fue especificación primero, con Spec Kit}
  \note{El mensaje en una frase: el esfuerzo se puso ANTES de generar
  código. Los prompts de /specify no fueron «hazme un sistema de reservas»,
  sino el alcance entero traducido a escenarios verificables. Por eso el
  código salió alineado con el documento del docente y no con lo que el
  modelo imaginó.}

  \begin{center}\footnotesize
    \id{/constitution} $\rightarrow$ \alert{\id{/specify}} $\rightarrow$
    \id{/plan} $\rightarrow$ \id{/tasks} $\rightarrow$ \id{/implement}
  \end{center}

  \vspace{2pt}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      {\footnotesize Antes de generar la primera línea de código, los prompts
      de \id{/specify} dejaron escrito:}
      \begin{itemize}\scriptsize
        \item \textbf{7 principios} de constitución, 2 de ellos no negociables.
        \item \textbf{48 requisitos funcionales} y \textbf{31 escenarios}
              \emph{Given / When / Then}.
        \item \textbf{10 criterios de éxito medibles}, no adjetivos.
        \item \textbf{4 contratos OpenAPI} (20 rutas), acordados antes del
              primer \emph{endpoint}.
        \item \textbf{134 tareas} derivadas del plan. Todas cerradas.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Qué compró ese esfuerzo}
        \scriptsize
        Cada regla de negocio se puede seguir desde el §3.4 del alcance hasta
        una clase y una prueba con nombre propio.\\[3pt]
        El agente no inventó alcance: lo que no estaba especificado, no se
        construyó.
      \end{block}
      {\scriptsize\color{textoSuave} $\approx$ 17\,500 palabras de
      especificación antes de la primera línea de código.}
    \end{column}
  \end{columns}
\end{frame}
